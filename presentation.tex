\begin{frame}{Áttekintés: minimális feszítőfa, csereél keresése}
  \small
  \begin{columns}[c]
    \column{0.55\textwidth}
      \begin{block}{Definíció:}
      Legyen $G = (V, E)$ egy irányítatlan, súlyozott gráf, ahol $n = |V|$ és $m = |E|$, és minden $e \in E$ élhez tartozik egy $w(e)$ súly.
      Egy minimális feszítőfa $T = MST(G)$ a gráf olyan $n-1$ élből álló részhalmaza, amely összeköti az összes csúcsot, és az élek összsúlya minimális.
      \end{block}
      \begin{block}{Definíció:}
        A csereél az a legkisebb súlyú él, amely újra összeköti a minimális feszítőfát (egy minimális feszítőfabeli él törlése után).
      \end{block}
      \vspace{6pt}
    \column{0.40\textwidth}
      \begin{center}
        \includegraphics[width=\linewidth]{./images/MST_1.png}\\[4pt]
        {\tiny Kép forrás: Wikipédia}
      \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Az algoritmus célja és eredménye (1.)}
\small
\begin{itemize}
  \begin{block}{Cél:}
    Keressük meg a minimális költségű csereélét minden minimális feszítőfabeli élhez.
  \end{block}
  \item Az algoritmus képes megtalálni ezeket, ha a nem min feszfabeli élek élsúly szerint növekvő sorrendben rendezve megvannak határozva:
  \begin{itemize}
    \item \textbf{Idő- és tárbonyolultság: $ \bigO ( m + n)$} \\
    \vspace{0.5em}
    (n: csúcsok száma, m: élek száma)
  \end{itemize}
  \item A cikk előtt lehetett tudni, hogy ez elméletileg lehetőséges, de ez az első algoritmikus megoldás erre.
\end{itemize}
\end{frame}

\begin{frame}{Az algoritmus célja és eredménye (2.)}
\small
\begin{itemize}
 \item Megjegyzés: a nem min.feszfabeli éleket rendezve kapjuk meg, ha az MST-t a Kruskal algoritmussal keressük meg.
  \begin{itemize}
    \item Ha nem a Kruskal: a nem MST éleket $\bigO (m)$ időbonyolultsággal rendezhetjük, ha ismerünk egy felső korlátot a lehetséges értékekre. (pl. Radix sort)
  \end{itemize}
  \item A cikk előtti legjobb eredmény egy $ \bigO (m \alpha (m, n))$ időbonyolultságú algoritmus volt Tarjan-tól 1979-ből. (megj.: $\alpha (m, n)$: Inverse Ackermann's function) 
\end{itemize}
\end{frame}

\begin{frame}{Legfontosabb él (Most Vital Edge)}
\small
\begin{itemize}
  \item \textbf{Definíció:} Az az él, amelynek eltávolítása a legnagyobb növekedést okozza az MST (minimális feszítőfa) súlyában.
  \item \textbf{Megjegyzés:} Ha $G$-ben van híd, a legfontosabb él nincs értelmezve.
  \item \textbf{Kulcsállítás:} A legfontosabb él az MST-ben van.
  \item \textbf{Az algoritmus hozzájárulása:}
    \begin{itemize}
      \item A csereélek meghatározása után a legfontosabb él kiválasztható $ \bigO (n)$ időben. Csak meg kell keresni, hogy melyik MST-beli él súlyának a különbsége legnagyobb a csereélének súlyával.
      \item Így: a legfontosabb él kiszámítható $ \bigO (m + n)$ időben, ha adott az eredeti MST és a nem MST élek rendezve.
    \end{itemize}
\end{itemize}
\end{frame}
