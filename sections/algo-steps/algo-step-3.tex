\begin{frame}{Példa lefutás - $(e,g)$ fundamentális köre}
\begin{columns}[T]
  \begin{column}{0.46\textwidth}
    \small
    \textbf{Fundamentális kör:}
    \begin{itemize}
      \item A nem-fa él $(e,g)$ az MST-ben a
            $g \rightarrow h \rightarrow e \rightarrow g$ kört indukálja.
      \item \texttt{PathLabel(g, e, (e,g))}:
        \begin{itemize}
          \item induló csúcs: $v \leftarrow g$,
          \item $\texttt{find(g)} = g$  $\Rightarrow$ még nincs tömörítve,
          \item $R(g,h) = (e,g)$, majd \texttt{link(g)}.
        \end{itemize}
      \item A $(g,h)$ MST-él elsőként kap csereélt.
    \end{itemize}
  \end{column}
