  \begin{column}{0.46\textwidth}
    \small
    \textbf{Második iteráció:}
    \begin{itemize}
      \item \texttt{link(g)} után \texttt{find(g)} már $h$-t adja vissza:
            a $(g,h)$ szakasz tömörült.
      \item Következő csúcs: $v \leftarrow h$, itt $\texttt{find(h)} = h$,
            ezért $R(h,e) = (e,g)$, majd \texttt{link(h)}.
      \item A teljes $(g,h,e)$ út egyetlen képviselő alá kerül.
    \end{itemize}
    \vspace{0.5em}
    \textbf{Következmény:}
    \begin{itemize}
      \item Későbbi nem-fa élek futásánál a \texttt{find} hívások
            \alert{átugorják} ezt az utat.
      \item Minden MST él legfeljebb egyszer kerül bejárásra.
    \end{itemize}
  \end{column}
